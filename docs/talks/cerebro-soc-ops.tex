\begin{frame}{Reality Check: Life on SecOps On-Call}
  \begin{itemize}
    \item Findings live in five tools, none agree; analysts lose the first hour reconciling truth.
    \item Evidence requests arrive during active incidents, forcing screenshot theater.
    \item Leadership needs answers now: "Are we compromised?" and "Can auditors sign off?"
  \end{itemize}
\end{frame}

\begin{frame}{What Cerebro Promises}
  \begin{itemize}
    \item \textbf{One nervous system}: every interface hits the same append-only store, so "source of truth" is literal.
    \item \textbf{Answers on demand}: graph + SQL over live data replaces stale exports and ad-hoc spreadsheets.
    \item \textbf{Guardrailed automation}: agents accelerate response without bypassing approvals or audit trails.
  \end{itemize}
\end{frame}

\section{The First Hour}

\begin{frame}{Before vs. With Cerebro: 08:00 Pager}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \textbf{Old World}
      \begin{itemize}
        \item Refresh five consoles, Slack the IAM team, hope CSV exports finish.
        \item Manual notes in docs, no audit trail for leadership checks.
        \item Every exec update is guesswork stitched from stale exports.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{With Cerebro}
      \begin{itemize}
        \item Spin up scoped agent session; agents, CLI, UI share identical findings.
        \item Every question, tool call, and decision lands in \texttt{agent\_audit\_events}.
        \item Query engine + timelines feed exec-ready answers in minutes.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{How We Kick Off the Day}
  \begin{itemize}
    \item \textbf{Problem}: Analysts waste time validating whether alerts are real.
    \item \textbf{Move}: On-call runs \texttt{make dev} locally, launches agent session locked to the hot org.
    \item \textbf{Outcome}: Within three minutes, they have an annotated list of high-risk findings with mapped controls.
  \end{itemize}
\end{frame}

\begin{frame}{Analyst POV}

\end{frame}

\section{Telemetry That Matters}

\begin{frame}{Why Unified Collection Matters}
  \begin{itemize}
    \item Without it, asset inventories drift and responders chase ghosts.
    \item Cerebro's collector sweeps resources, principals, configs, and IAM edges in one motion.
    \item Identity stitching turns SaaS fragments into human-readable owners, so escalations hit the right inbox.
  \end{itemize}
\end{frame}

\begin{frame}{Story: New Acquisition, Zero Visibility}
  \begin{itemize}
    \item \textbf{Trigger}: M\&A team drops a GitHub org on Friday night.
    \item \textbf{Cerebro Move}: Collector job runs in background; account inventories populate, configs hashed, alerts normalized.
    \item \textbf{Outcome}: Monday 09:00 briefing includes exposure counts, owners, and evidence bundles—no heroic spreadsheeting.
  \end{itemize}
\end{frame}

\begin{frame}{The Query Engine in Human Terms}
  \begin{itemize}
    \item Analysts phrase questions like \emph{"Show MFA gaps for prod admins"}; the engine answers in seconds.
    \item Dashboards, agent responses, and compliance reports all reuse the same SQL call—consistency by design.
    \item Evidence generation pulls straight from those queries, so auditors get signed artifacts instead of screenshots.
  \end{itemize}
\end{frame}

\section{Incident Response}

\begin{frame}{Scenario: Leaked Token Midnight Wednesday}
  \begin{itemize}
    \item Agent session seeded with incident ID auto-loads prior context and open findings.
    \item Responder asks \emph{"Rebuild the compromise timeline"}; \texttt{timeline} tool reconstructs actions with timestamps and actors.
    \item Leadership call gets facts: blast radius, affected principals, mitigations underway.
  \end{itemize}
\end{frame}

\begin{frame}{Why Agents Stay Trustworthy}
  \begin{itemize}
    \item ToolExecutor enforces dry-run and CEL policies; destructive moves require explicit approval.
    \item Every tool invocation logs who asked, what context was used, and how long it took—perfect for post-incident reviews.
    \item If an agent hallucinates, analysts override once and the audit record captures the correction for follow-up.
  \end{itemize}
\end{frame}

\begin{frame}{Evidence Without Panic}
  \begin{itemize}
    \item WORM evidence bundles package SQL, configs, and screenshots with signatures.
    \item Chain-of-custody lives inside the bundle; no more emailing ZIPs around.
    \item Same artifacts satisfy incident retros, regulator questions, and customer assurances.
  \end{itemize}
\end{frame}

\section{Compliance Meets Operations}

\begin{frame}{Why Finance and Audit Care}
  \begin{itemize}
    \item Control tests run continuously; failed tests route straight into the remediation queue SecOps already works.
    \item Policy attestations, device posture, and technical findings share one dashboard—no duplicate status meetings.
    \item When auditors ask "Prove MFA is enforced," we hand over the signed evidence bundle generated during the incident.
  \end{itemize}
\end{frame}

\begin{frame}{One Platform, Two Conversations}
  \begin{itemize}
    \item SOC leaders talk in risk narratives; finance speaks SOC 2 controls—Cerebro outputs both from the same data.
    \item Time-to-audit-packet dropped from days to hours once evidence bundling shipped.
    \item We now justify budget by pointing to avoided fines and measurable response time cuts.
  \end{itemize}
\end{frame}

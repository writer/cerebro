\section{Your Daily Workflows}

\begin{frame}{Monday Morning: Weekly Control Review}
  \textbf{What You Need to Know}: Are we still in compliance?
  
  \vspace{0.3cm}
  
  \textbf{Where to Look}:
  \begin{itemize}
    \item Dashboard at \texttt{/analytics/executive} shows overall compliance posture
    \item Control test results at \texttt{/compliance/controls} with pass/fail status
    \item Failed controls highlighted with remediation owners assigned
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{What to Do}:
  \begin{itemize}
    \item Review any newly failed controls—are they real issues or false positives?
    \item Follow up with assigned owners on overdue remediation tasks
    \item Note trends: are specific control families degrading?
  \end{itemize}
\end{frame}

\begin{frame}{Daily Task: Reviewing Control Exceptions}
  \textbf{Scenario}: A team requests an exception for a failed control
  
  \vspace{0.3cm}
  
  \textbf{Your Workflow}:
  \begin{enumerate}
    \item Navigate to \texttt{/compliance/controls} and find the failing control
    \item Review the control test details: what specifically is failing?
    \item Click "Create Exception" and document:
      \begin{itemize}
        \item Business justification
        \item Compensating controls in place
        \item Expiration date (30/60/90 days)
        \item Required approvals (your signature + CISO)
      \end{itemize}
    \item Exception is logged with timestamp and cannot be altered later
  \end{enumerate}
  
  \vspace{0.3cm}
  
  \textbf{Audit Value}: Auditors see documented risk acceptance with proper approval chain
\end{frame}

\begin{frame}{Mid-Week: Evidence Collection Check}
  \textbf{What You're Monitoring}: Is evidence flowing properly?
  
  \vspace{0.3cm}
  
  \textbf{Key Indicators at \texttt{/compliance/evidence}}:
  \begin{itemize}
    \item \textbf{Freshness}: Last collection timestamp per control
    \item \textbf{Coverage}: Which controls have current evidence vs. stale
    \item \textbf{Health}: Any collection failures requiring IT follow-up
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Common Issues}:
  \begin{itemize}
    \item Integration disconnected: coordinate with IT to reconnect (e.g., Okta, AWS)
    \item Manual evidence needed: assign task to control owner with due date
    \item Policy updates required: flag for next policy review cycle
  \end{itemize}
\end{frame}

\begin{frame}{Friday Afternoon: Policy Attestation Status}
  \textbf{Goal}: Ensure employees acknowledge required policies
  
  \vspace{0.3cm}
  
  \textbf{Your Dashboard View at \texttt{/compliance/policies}}:
  \begin{itemize}
    \item Overall attestation rate: 94\% of employees current
    \item Overdue attestations: 12 employees need to sign acceptable use policy
    \item Department breakdown: Engineering 98\%, Sales 87\%
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Actions}:
  \begin{itemize}
    \item Send reminder email to overdue employees (one-click from dashboard)
    \item Escalate to department managers if 7+ days overdue
    \item Review new hire attestation completion: should be 100\% within first week
  \end{itemize}
\end{frame}

\section{Audit Preparation}

\begin{frame}{Four Weeks Before Audit: Readiness Check}
  \textbf{Use the Audit Readiness Report at \texttt{/compliance/audit-readiness}}:
  
  \vspace{0.3cm}
  
  \begin{itemize}
    \item \textbf{Control coverage}: 87 of 90 SOC2 controls have current evidence
    \item \textbf{Evidence gaps}: 3 controls need manual evidence collection
    \item \textbf{Policy status}: All policies up-to-date, 96\% employee attestation
    \item \textbf{Exception review}: 4 active exceptions, all properly documented
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Pre-Audit Checklist}:
  \begin{itemize}
    \item Close out any overdue remediation tasks or document as exceptions
    \item Ensure all manual evidence is collected and uploaded
    \item Review control test history for any suspicious patterns
    \item Generate preliminary evidence bundle for internal review
  \end{itemize}
\end{frame}

\begin{frame}{During Audit: Request List Management}
  \textbf{Scenario}: Auditor sends request list with 45 items
  
  \vspace{0.3cm}
  
  \textbf{Cerebro Workflow}:
  \begin{enumerate}
    \item Upload request list to \texttt{/compliance/audit-workflow}
    \item System auto-matches requests to existing evidence (30 items matched)
    \item Remaining 15 items auto-assigned to control owners
    \item Track completion: 42/45 complete, 3 in progress
  \end{enumerate}
  
  \vspace{0.3cm}
  
  \textbf{Benefits}:
  \begin{itemize}
    \item No more "who's working on what?" confusion
    \item Evidence delivered in auditor-friendly format
    \item Chain of custody preserved for each artifact
    \item Typical audit response time: 2-3 days vs. 2 weeks
  \end{itemize}
\end{frame}

\begin{frame}{Evidence Bundle Generation}
  \textbf{What Auditors Receive}:
  
  \vspace{0.3cm}
  
  \textbf{Bundle Contents} (generated via \texttt{/compliance/evidence/export}):
  \begin{itemize}
    \item Manifest file listing all included evidence with timestamps
    \item Control test results with pass/fail status and screenshots
    \item Policy documents with version history and attestation records
    \item System configuration snapshots (IAM, security groups, MFA status)
    \item Digital signatures proving evidence hasn't been tampered with
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Format}: ZIP file with PDF report + JSON data for auditor's analysis tools
  
  \vspace{0.3cm}
  
  \textbf{Auditor Benefits}:
  \begin{itemize}
    \item Clear evidence organization matching their testing procedures
    \item Cryptographic verification of evidence integrity
    \item Point-in-time snapshots for any date in audit period
  \end{itemize}
\end{frame}

\section{Risk Management Scenarios}

\begin{frame}{Scenario: Executive Risk Briefing}
  \textbf{Request}: CFO needs compliance status for board meeting
  
  \vspace{0.3cm}
  
  \textbf{Your 5-Minute Workflow}:
  \begin{enumerate}
    \item Navigate to \texttt{/analytics/executive}
    \item Export executive summary showing:
      \begin{itemize}
        \item Overall compliance score: 94/100
        \item Control pass rate: 92\% (83 passing, 7 failing)
        \item Recent improvements: MFA compliance 78\% → 96\%
        \item Open risks: 2 high-priority findings under remediation
      \end{itemize}
    \item Add narrative context in provided template
    \item Export as board-ready PDF with charts
  \end{enumerate}
  
  \vspace{0.3cm}
  
  \textbf{Result}: Professional summary without technical jargon, audit-trail backed numbers
\end{frame}

\begin{frame}{Scenario: Vendor Risk Assessment}
  \textbf{Context}: Security team flags a critical vendor; GRC needs risk assessment
  
  \vspace{0.3cm}
  
  \textbf{Cerebro Helps You Understand}:
  \begin{itemize}
    \item \textbf{At \texttt{/security-center/vendors}}: vendor profile with risk rating
    \item \textbf{What data they access}: linked to data classification and systems
    \item \textbf{Compliance status}: SOC2/ISO27001 report status and expiration
    \item \textbf{Incident history}: any prior security events involving this vendor
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Your Assessment}:
  \begin{itemize}
    \item Document risk level with supporting evidence
    \item Assign compensating controls (e.g., enhanced monitoring)
    \item Set review date for next assessment
    \item All documentation auto-captured for audit trail
  \end{itemize}
\end{frame}

\begin{frame}{Scenario: Failed Control Investigation}
  \textbf{Alert}: Automated control test fails—"Privileged accounts without MFA"
  
  \vspace{0.3cm}
  
  \textbf{Your Investigation Path}:
  \begin{enumerate}
    \item Click alert to see control test details
    \item Review affected accounts: 3 admin accounts flagged
    \item Check if legitimate: are these service accounts or human users?
    \item Coordinate remediation:
      \begin{itemize}
        \item Human accounts: IT enables MFA within 24 hours
        \item Service accounts: document exception with compensating controls
      \end{itemize}
    \item Re-run control test to confirm remediation
  \end{enumerate}
  
  \vspace{0.3cm}
  
  \textbf{Audit Value}: Complete timeline of detection → investigation → resolution
\end{frame}

\section{Common Questions}

\begin{frame}{How Often Should I Check Cerebro?}
  \textbf{Daily}:
  \begin{itemize}
    \item Quick dashboard review (2 minutes): any new critical failures?
    \item Policy attestation reminders: any urgent overdue items?
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Weekly}:
  \begin{itemize}
    \item Control test results: trends and patterns
    \item Evidence collection health: any integration issues?
    \item Exception review: approaching expiration dates?
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Monthly}:
  \begin{itemize}
    \item Executive summary report for leadership
    \item Control effectiveness review with security team
    \item Policy review cycle: any updates needed?
  \end{itemize}
\end{frame}

\begin{frame}{What If I Don't Understand a Technical Finding?}
  \textbf{Translation Help Built-In}:
  
  \vspace{0.3cm}
  
  \begin{itemize}
    \item Each control has \textbf{"Business Impact"} section explaining why it matters
    \item \textbf{"Recommended Action"} provides non-technical remediation guidance
    \item \textbf{Ask the agent}: conversational interface at \texttt{/agents/chat}
      \begin{itemize}
        \item Ask: "Explain this IAM finding in simple terms"
        \item Agent provides context without requiring security expertise
      \end{itemize}
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{When to Escalate}:
  \begin{itemize}
    \item You're unsure if a control failure is acceptable: consult CISO
    \item Remediation requires technical work: assign to security engineering
    \item Auditor asks technical questions: coordinate with security team
  \end{itemize}
\end{frame}

\begin{frame}{How Do I Trust the Data?}
  \textbf{Built-In Integrity Guarantees}:
  
  \vspace{0.3cm}
  
  \begin{itemize}
    \item \textbf{Immutable audit logs}: once recorded, evidence cannot be altered
    \item \textbf{Cryptographic signatures}: each evidence artifact is digitally signed
    \item \textbf{Chain of custody}: timestamp and source system for every record
    \item \textbf{Point-in-time queries}: retrieve exact state from any past date
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Auditor Verification}:
  \begin{itemize}
    \item Auditors can verify signatures independently
    \item SQL query engine allows auditors to re-run tests themselves
    \item Evidence bundles include manifest with checksums
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Your Confidence}: When you present evidence, it's legally defensible
\end{frame}

\section{Quick Start}

\begin{frame}[fragile]{Python: 30-Second Setup}
\textbf{Install}:
\begin{lstlisting}[language=bash]
uv pip install -e /path/to/cerebro
# Or add to pyproject.toml dependencies
\end{lstlisting}

\textbf{Get a database session}:
\begin{lstlisting}[language=Python]
from cerebro.core.database import async_session_factory

async with async_session_factory() as db:
    # Use SDK here
    pass
\end{lstlisting}

\textbf{Import what you need}:
\begin{lstlisting}[language=Python]
from cerebro_sdk import (
    AuthSession,      # Login and tokens
    AgentManager,     # Agent sessions
    FindingService,   # Security findings
    AgentReviewManager,  # Review queue
)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{TypeScript: 30-Second Setup}
\textbf{Install}:
\begin{lstlisting}[language=bash]
npm install @writerai/cerebro-sdk
\end{lstlisting}

\textbf{Initialize client}:
\begin{lstlisting}[language=Java]
import { CerebroSDK } from "@writerai/cerebro-sdk";

const sdk = new CerebroSDK({
  baseUrl: "https://cerebro.writer.com",
  async getAccessToken() {
    return await yourAuthService.getToken();
  }
});
\end{lstlisting}

\textbf{Use the clients}:
\begin{lstlisting}[language=Java]
// All functionality under sdk.{module}
sdk.agents.createSession(...)
sdk.findings.list(...)
sdk.securityCenter.listVendors(...)
\end{lstlisting}
\end{frame}

\section{Agent Operations}

\begin{frame}[fragile]{Create an Agent Session}
\textbf{Use Case}: Start a new agent conversation

\textbf{Python}:
\begin{lstlisting}[language=Python]
from cerebro_sdk import AgentManager

agents = AgentManager(db)
session = await agents.create_session(
    org_id=UUID("your-org-id"),
    agent_type="security_analyst",  # or incident_responder
    created_by="your-email@writer.com",
    context={"finding_ids": ["finding-123"]},
    title="Investigate critical finding"
)

# Result: AgentSessionRecord
print(session.session_id)  # UUID of new session
print(session.is_active)   # True
\end{lstlisting}

\textbf{TypeScript}:
\begin{lstlisting}[language=Java]
const session = await sdk.agents.createSession({
  orgId: "your-org-id",
  agentType: "security_analyst",
  context: { findingIds: ["finding-123"] }
});
\end{lstlisting}
\end{frame}

\section{Common Patterns}

\begin{frame}[fragile]{Pattern: Automated Finding Triage}
\textbf{Use Case}: Auto-create agent sessions for critical findings

\begin{lstlisting}[language=Python]
from cerebro_sdk import FindingService, AgentManager

async def auto_triage_critical(org_id):
    async with async_session_factory() as db:
        findings_svc = FindingService(db)
        agents = AgentManager(db)
        
        # Get new critical findings
        findings = await findings_svc.list_findings(
            org_id=org_id,
            severity="critical",
            status="open",
            limit=10
        )
        
        for finding in findings:
            # Create agent session
            session = await agents.create_session(
                org_id=org_id,
                agent_type="security_analyst",
                created_by="automation@writer.com",
                context={"finding_ids": [str(finding.finding_id)]},
                title=f"Triage: {finding.title}"
            )
            
            # Queue initial analysis
            await agents.add_message(
                session_id=session.session_id,
                role="user",
                content={"text": "Analyze blast radius and priority"}
            )
            
            print(f"Created session {session.session_id} for {finding.finding_id}")
\end{lstlisting}
\end{frame}

        # Get SLA at-risk
        at_risk = await review.list_tasks_sla_at_risk(
            org_id=org_id,
            threshold_hours=24
        )
        
        # Get summary
        summary = await review.get_queue_summary(org_id)
        
        return {
            "pending_count": len(pending),
            "at_risk_count": len(at_risk),
            "avg_age_hours": summary.avg_age_hours,
            "oldest_task_age": summary.oldest_task_age,
            "by_priority": {
                "high": len([t for t in pending if t.priority == "high"]),
                "medium": len([t for t in pending if t.priority == "medium"]),
                "low": len([t for t in pending if t.priority == "low"])
            }
        }

# Usage
metrics = await get_queue_metrics(org_id)
print(f"Pending: {metrics['pending_count']}")

import { CerebroSDK, assessVendorHealth, 
         summarizeVendorPortfolio } from "@writerai/cerebro-sdk";

async function generateComplianceReport(orgId: string) {
  const sdk = new CerebroSDK({...});

  return {

    byCategory: portfolio.byCategory,

}

from cerebro_sdk import AgentToolingManager
from cerebro.agents.models import ToolInvocationStatus

async def audit_tool_usage(org_id, days=7):
    async with async_session_factory() as db:
        tooling = AgentToolingManager(db)
        
        # Get all tool calls
        invocations = await tooling.list_invocations(
            org_id=org_id,
            limit=1000
        )
        
        # Group by tool
        by_tool = {}
        for inv in invocations:
            if inv.tool_name not in by_tool:
                by_tool[inv.tool_name] = {
                    "total": 0, "success": 0, "failed": 0,
                    "total_duration": 0
                }
            
            by_tool[inv.tool_name]["total"] += 1
            if inv.status == ToolInvocationStatus.SUCCESS:
                by_tool[inv.tool_name]["success"] += 1
            elif inv.status == ToolInvocationStatus.ERROR:
                by_tool[inv.tool_name]["failed"] += 1
            if inv.duration_seconds:
                by_tool[inv.tool_name]["total_duration"] += inv.duration_seconds
        
        # Print report
        for tool, stats in by_tool.items():
            success_rate = (stats["success"] / stats["total"]) * 100
            avg_duration = stats["total_duration"] / stats["total"]
            print(f"{tool}:")
            print(f"  Calls: {stats['total']}")
            print(f"  Success rate: {success_rate:.1f}%")
            print(f"  Avg duration: {avg_duration:.2f}s")
\end{lstlisting}

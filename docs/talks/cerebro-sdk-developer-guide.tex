\section{Introduction}

\begin{frame}{Why Use the Cerebro SDK?}
  \textbf{The Problem}:
  \begin{itemize}
    \item Ad-hoc REST calls scattered across services
    \item No type safety, manual JSON parsing
    \item Duplicate pagination and error handling logic
    \item Difficult to test without hitting live APIs
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{The SDK Solution}:
  \begin{itemize}
    \item Typed interfaces for all Cerebro operations
    \item Consistent patterns for async workflows
    \item Built-in observability and error handling
    \item Test adapters for offline development
  \end{itemize}
\end{frame}

\begin{frame}{Two SDKs, One Platform}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \textbf{TypeScript SDK} (\texttt{sdk/ts})
      \begin{itemize}
        \item Frontend integrations
        \item Browser-based tools
        \item Partner APIs
        \item Real-time streaming
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Python SDK} (\texttt{src/cerebro\_sdk})
      \begin{itemize}
        \item Backend automation
        \item Compliance tooling
        \item Async workflows
        \item Database-backed operations
      \end{itemize}
    \end{column}
  \end{columns}
  
  \vspace{0.5cm}
  
  Both SDKs share:
  \begin{itemize}
    \item Identical domain models (agents, findings, evidence)
    \item Same API surface and semantics
    \item Parallel feature development
  \end{itemize}
\end{frame}

\begin{frame}{Core Capabilities}
  \textbf{What the SDK Provides}:
  \begin{itemize}
    \item \textbf{Agent Operations}: sessions, memory, review workflows, tool execution
    \item \textbf{Security Intelligence}: findings, analytics, runtime health
    \item \textbf{Compliance}: control testing, evidence management, audit workflows
    \item \textbf{Integration Management}: provider sync, coverage tracking
    \item \textbf{Observability}: structured logging, Prometheus metrics
  \end{itemize}
\end{frame}

\section{Production Patterns}

\begin{frame}{Error Handling Strategy}
  \textbf{SDK Exceptions Hierarchy}:
  \begin{itemize}
    \item \texttt{AgentSDKError} (base for all SDK errors)
    \begin{itemize}
      \item \texttt{AgentNotFoundError}: resource doesn't exist
      \item \texttt{AgentInvalidStatusError}: invalid state transition
      \item \texttt{AgentValidationError}: input validation failed
    \end{itemize}
    \item \texttt{HttpError} (TypeScript): HTTP request failures
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Recommended Pattern}:
  \begin{itemize}
    \item Catch specific exceptions for known failure modes
    \item Log \texttt{AgentSDKError} base class for unexpected errors
    \item Include context (org\_id, session\_id) in error logs
    \item Retry on transient network errors, not validation errors
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Retry Pattern}
\textbf{TypeScript with Exponential Backoff}:
\begin{lstlisting}[language=Java]
import { CerebroSDK, HttpError } from "@writerai/cerebro-sdk";

async function withRetry<T>(
  fn: () => Promise<T>,
  maxAttempts = 3
): Promise<T> {
  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
    try {
      return await fn();
    } catch (error) {
      if (error instanceof HttpError && 
          error.status >= 500 && 
          attempt < maxAttempts) {
        const delay = Math.pow(2, attempt) * 1000;
        await new Promise(r => setTimeout(r, delay));
        continue;
      }
      throw error;
    }
  }
  throw new Error("Unreachable");
}

const sessions = await withRetry(
  () => sdk.agents.listSessions({ orgId })
);
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Connection Pool Management}

from contextlib import asynccontextmanager

\begin{frame}{Rate Limiting and Throttling}
  \textbf{SDK Built-in Protections}:
  \begin{itemize}
    \item Respects HTTP 429 responses with Retry-After headers
    \item Configurable timeout per request (default: 30s)
    \item Connection pooling prevents overwhelming backend
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Application-Level Strategies}:
  \begin{itemize}
    \item Use pagination limits to control batch sizes
    \item Implement job queues for bulk operations
    \item Cache frequently accessed data (findings, org metadata)
    \item Monitor SDK metrics for throttling signals
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Recommended Limits}:
  \begin{itemize}
    \item Findings/sessions: 100 per page
    \item Review tasks: 50 per page
    \item Concurrent API calls: 10 per service
  \end{itemize}
\end{frame}

\begin{frame}{Configuration Management}
  \textbf{Python: Settings Proxy Pattern}
  \begin{itemize}
    \item Cache settings in-memory with automatic refresh
    \item Environment variable overrides
    \item Safe defaults for development vs. production
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{TypeScript: Environment-Aware Setup}
  \begin{itemize}
    \item Different base URLs per environment
    \item Token providers vary by runtime (browser vs. Node.js)
    \item Custom headers for service identification
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Secrets Management}:
  \begin{itemize}
    \item Never hardcode API keys or passwords
    \item Use environment variables or secret managers
    \item Rotate tokens regularly, SDK supports async token refresh
  \end{itemize}

  \textbf{Inspect Raw API Responses}:
  \begin{itemize}
    \item Python: Check \texttt{AsyncSession.execute()} SQL output
    \item TypeScript: Use browser DevTools Network tab
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Test in Isolation}:
  \begin{itemize}
    \item Use in-memory adapters to rule out API issues
    \item Try equivalent curl commands to verify API behavior
    \item Check Cerebro logs for server-side errors
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Version Compatibility}:
  \begin{itemize}
    \item SDK version must match API version
    \item Check \texttt{CHANGELOG.md} for breaking changes
  \end{itemize}
\end{frame}

\section{Best Practices Summary}

\begin{frame}{Development Best Practices}
  \textbf{Code Organization}:
  \begin{itemize}
    \item Create service wrappers around SDK clients
    \item Don't pass raw database sessions to business logic
    \item Use dependency injection for testability
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Error Handling}:
  \begin{itemize}
    \item Always catch SDK-specific exceptions
    \item Log errors with sufficient context
    \item Don't retry validation errors, only transient failures
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Performance}:
  \begin{itemize}
    \item Use pagination for large result sets
    \item Cache frequently accessed data
    \item Monitor database connection pool usage
  \end{itemize}
\end{frame}

\begin{frame}{Production Best Practices}
  \textbf{Observability}:
  \begin{itemize}
    \item Emit metrics for SDK operations (counters, histograms)
    \item Log request IDs for distributed tracing
    \item Alert on error rate spikes
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Security}:
  \begin{itemize}
    \item Use short-lived access tokens
    \item Rotate credentials regularly
    \item Validate org\_id scope before operations
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Reliability}:
  \begin{itemize}
    \item Implement retry logic with exponential backoff
    \item Set appropriate timeouts (30-60s for most operations)
    \item Have fallback behavior for degraded API states
  \end{itemize}
\end{frame}

\begin{frame}{Testing Best Practices}
  \textbf{Unit Testing}:
  \begin{itemize}
    \item Use in-memory adapters, not live APIs
    \item Test edge cases and error paths
    \item Keep tests fast (< 100ms per test)
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Integration Testing}:
  \begin{itemize}
    \item Test against local Cerebro instance
    \item Clean up test data after runs
    \item Use dedicated test organizations
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Contract Testing}:
  \begin{itemize}
    \item Verify SDK matches API contracts
    \item Test serialization/deserialization
    \item Catch schema drift early
  \end{itemize}
\end{frame}

\section{Orientation}

\begin{frame}{What is Cerebro?}
  \begin{itemize}
    \item Cerebro is Writer's security operations nervous system: agents, telemetry, and automation.
    \item SDK delivers a typed interface for product, partner, and internal teams to orchestrate Cerebro.
    \item Goals: minimize bespoke REST calls, codify policies, and enable rapid prototyping with guardrails.
  \end{itemize}
\end{frame}

\begin{frame}{Design Principles}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \textbf{Predictable shapes}: canonical mappers in \texttt{clients/*} enforce typed records.
        \item \textbf{Adapter-first}: HTTP and in-memory adapters keep testing and offline demos trivial.
        \item \textbf{Observability built-in}: logging, counters, and histograms ship out-of-the-box.
        \item \textbf{Environment aware}: pluggable settings loader and token providers support any runtime.
      \end{itemize}
    \end{column}
    \begin{column}{0.38\textwidth}
      \begin{block}{Fast facts}
        \begin{itemize}
          \item TypeScript + Vitest + Changesets
          \item Zero-dependency runtime (only \texttt{fetch})
          \item Distributed via \texttt{@writerai/cerebro-sdk}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Module Map}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item \textbf{Core}: \texttt{httpClient.ts}, \texttt{sdk.ts}, \texttt{config.ts}, \texttt{telemetry.ts}
        \item \textbf{Domain clients}: agents, analytics, findings, securityCenter, integrations, organizations
        \item \textbf{Agent services}: analytics, review exporter, security summarizers
        \item \textbf{Testing harness}: deterministic adapters in \texttt{src/testing}
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Packaging}
        \begin{itemize}
          \item Entry point re-exports everything via \texttt{src/index.ts}
          \item \texttt{dist/} yields ESM bundles, tree-shake friendly
          \item Changeset guard enforces semantic releases
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Core Mechanics}

\begin{frame}{HTTP Foundation}
  \begin{itemize}
    \item \texttt{HttpClient} normalises base URLs, search params, JSON bodies, and error handling (\texttt{HttpError}).
    \item Token acquisition pluggable via \texttt{getAccessToken}; supports async refresh before each call.
    \item Portable: supply \texttt{fetch} for Node.js, Service Workers, or test doubles.
    \item Shared pagination helpers deliver cursor + offset flows with \texttt{CursorPage}, \texttt{PageRequest}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Bootstrapping the SDK}
  \begin{lstlisting}[language=Java]
import { CerebroSDK } from "@writerai/cerebro-sdk";

const sdk = new CerebroSDK({
  baseUrl: process.env.CEREBRO_BASE_URL!,
  async getAccessToken() {
    return authClient.getToken();
  },
  defaultHeaders: { "x-writer-surface": "brownbag-demo" },
});

const { sessions } = await sdk.agents.listSessions({
  agentType: "security_investigator",
  limit: 5,
});
\end{lstlisting}
\end{frame}

\begin{frame}{Configuration and Telemetry}
  \begin{itemize}
    \item \texttt{configureSettingsLoader} builds cached settings proxies for runtime toggles.
    \item Logger factory exposes leveled output, JSON formatting, and custom sinks for SIEM forwarding.
    \item Metrics helpers (counters + histograms) instrument agent workflows without Prometheus dependencies.
    \item \textbf{Pattern}: wire counters per tool, histogram around \texttt{AgentToolingClient.summarizeInvocations}.
  \end{itemize}
\end{frame}

\section{Agent Operations}

\begin{frame}{Agents Client Overview}
  \begin{itemize}
    \item Review queue APIs: list, paginate, bulk update, resolve, SLA summaries, notifications.
    \item Session lifecycle: create, fetch transcript + tool invocations, stream messages, manage memory.
    \item Workflow engine: list templates, evaluate triggers, simulate policy expressions.
    \item Analytics hooks: runtime events plus summaries for tool latency, failure signatures.
  \end{itemize}
\end{frame}

\begin{frame}{Agent Use Cases}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{block}{Human-in-the-loop Triage}
        \begin{itemize}
          \item Auto-queue tasks via \texttt{bulkUpdateReviewTasks} with prioritization knobs.
          \item Surface SLA risk early (\texttt{listReviewTasksSlaAtRisk}).
          \item Pipe notifications to PagerDuty / Slack using \texttt{AgentNotificationsClient}.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Autonomous Response}
        \begin{itemize}
          \item Generate workflow drafts via \texttt{evaluateWorkflows} using contextual metadata.
          \item Simulate policy CEL expressions before shipping to production.
          \item Replay memory slices for retros using \texttt{listSessionMemoryEntries}.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Tooling, Tickets, and Tasks}
  \begin{itemize}
    \item \textbf{AgentToolingClient}: create, update, and summarize tool invocations; bridge approvals with humans.
    \item \textbf{TasksClient}: enqueue backend jobs, poll statuses, and revoke runaway Celery tasks.
    \item \textbf{AgentTicketsClient}: open or resolve escalation tickets, reconcile external IDs.
    \item \textbf{Pattern}: use in-memory adapters for integration tests, swap to HTTP in production.
  \end{itemize}
\end{frame}

\section{Security Intelligence}

\begin{frame}{Findings and Analytics}
  \begin{itemize}
    \item Findings API exposes status, severity, provider filters, plus cursor pagination for large tenants.
    \item Analytics runtime health aggregates events, warnings, and metadata snapshots per runtime.
    \item Combine with agent analytics to build near-real-time detection dashboards.
    \item Use \texttt{CursorPage} helpers to stream findings into data warehouses without duplicate logic.
  \end{itemize}
\end{frame}

\begin{frame}{Security Center Insights}
  \begin{itemize}
    \item Register vendors and customers, capturing risk and lifecycle metadata via structured envelopes.
    \item Overview endpoint synthesizes metrics, upcoming compliance expirations, and submission queues.
    \item Vendor/customer listings expose normalized risk scoring for executive rollups.
    \item Metadata helpers (\texttt{toVendorMetadataEnvelope}) guard against malformed partner payloads.
  \end{itemize}
\end{frame}

\begin{frame}{Integrations and Organizations}
  \begin{itemize}
    \item Integration coverage tracks scope health, provider status, and last-success timestamps.
    \item Organization client delivers tenant discovery for multi-org deployments.
    \item Combine with security center to coordinate risk reviews per org + integration tuple.
    \item Auth client handles login + current-user introspection for management consoles.
  \end{itemize}
\end{frame}

\section{Developer Experience}

\begin{frame}{Testing and Simulation}
  \begin{itemize}
    \item Every client ships an in-memory adapter mirroring HTTP contracts for unit tests.
    \item \texttt{src/testing} provides canned analytics fixtures for replaying runtime events.
    \item Vitest suites cover adapters, ensuring consistent mapping semantics across transports.
    \item Pattern: write contract tests once against adapter interface, then reuse for HTTP + mocks.
  \end{itemize}
\end{frame}

\begin{frame}{Operational Best Practices}
  \begin{itemize}
    \item Centralize base URLs and tokens with the settings proxy; refresh via \texttt{refreshSettings} on rotation.
    \item Wrap API calls with telemetry timers to trend latency (e.g. policy simulations, workflow eval).
    \item Leverage pagination consistently; never assume finite review tasks or findings per org.
    \item On errors, bubble \texttt{HttpError.body} into observability stack for rapid diagnosis.
  \end{itemize}
\end{frame}

\begin{frame}{Rollout Checklist}
  \begin{itemize}
    \item Validate scopes: align agent + tooling permissions before enabling auto-remediation flows.
    \item Harden auth: prefer short-lived tokens, rotate secrets, and enforce CSRF tokens for browsers.
    \item Instrument counters before GA to benchmark review queue throughput.
    \item Document fallback plans: in-memory adapters double as chaos drills when APIs degrade.
  \end{itemize}
\end{frame}

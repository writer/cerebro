\section{Recon}

\begin{frame}{Repo Recon}
  \begin{itemize}
    \item Stack: FastAPI, Typer CLI, PostgreSQL, Redis, OpenAI/Anthropic agents.
    \item Data spine: append-only audit store with graph + SQL analytics.
    \item Key dirs: \texttt{src/cerebro/api}, \texttt{agents}, \texttt{core}, \texttt{collectors}, \texttt{rules}.
    \item Attack surface: REST API on 8000, web UI on 3000, CLI exec via agent runtimes.
  \end{itemize}
\end{frame}

\begin{frame}{Telemetry Topology}
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item Findings pipeline streams cloud misconfigs into immutable tables.
        \item CEL rule engine drives automated verdicts and remediation.
        \item Agents pipeline blends memory, skill-based routing, and tool execution.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Implication}
        Every control plane action leaves breadcrumbs; adversaries need to bend them, not break them.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Attack Surface}

\begin{frame}{Enumerating Control Points}
  \begin{itemize}
    \item API: FastAPI routers in \texttt{src/cerebro/api} expose ingestion and query endpoints.
    \item CLI: Typer commands orchestrate database interactions and agent sessions.
    \item Agents: tool latency metrics in \texttt{agents/tool\_stats.py} drive adaptive ordering.
    \item Memory: decay-aware embeddings with configurable pruning thresholds.
  \end{itemize}
\end{frame}

\begin{frame}{Privilege Choke Points}
  \begin{itemize}
    \item \textbf{Review queue}: promotes risky actions when human approves; target the workflow.
    \item \textbf{Skill routing}: biases runtime selection; poisoning telemetry could reroute to weaker models.
    \item \textbf{Finding normalization}: shape data to trigger arbitrary remediation hooks.
    \item \textbf{Graph engine}: SQL + graph queries without ETL, ripe for inference attacks.
  \end{itemize}
\end{frame}

\section{Weaponization}

\begin{frame}{Initial Access Play}
  \begin{itemize}
    \item Abuse default credentials \texttt{admin/admin123!} in lab deployments.
    \item Replay agent memory exports to harvest sensitive operational intel.
    \item Inject CEL rules to exfiltrate findings during remediation cycles.
  \end{itemize}
\end{frame}

\begin{frame}{Living off the Land}
  \begin{itemize}
    \item Leverage \texttt{cerebro query} to enumerate cloud posture blind spots.
    \item Use append-only audit trails to mask footprint with noise (log flooding).
    \item Ride adaptive tool ordering to starve defensive skills, forcing fail-open paths.
  \end{itemize}
\end{frame}

\section{Impact}

\begin{frame}{Attack Objectives}
  \begin{itemize}
    \item Own the telemetry: manipulate Prometheus counters and OTLP spans to mislead responders.
    \item Persist in \texttt{agents} skill registry to hijack future investigations.
    \item Weaponize compliance automation to rubber-stamp hostile infrastructure changes.
  \end{itemize}
\end{frame}

\begin{frame}{Signals of Compromise}
  \begin{itemize}
    \item Unusual review queue approvals for destructive agent actions.
    \item Memory decay thresholds suddenly increasing per org.
    \item Graph queries generating high-cardinality results without corresponding findings.
  \end{itemize}
\end{frame}

\section{Counterplay}

\begin{frame}{Defensive Recommendations}
  \begin{itemize}
    \item Rotate lab credentials and enforce MFA on the CLI.
    \item Instrument CEL policies with canaries to detect tampering.
    \item Enforce signed agent tools; validate telemetry before routing decisions.
    \item Monitor audit trail volume anomalies and throttle untrusted clients.
  \end{itemize}
\end{frame}

\begin{frame}{Call to Action}
  \begin{itemize}
    \item Threat model agent workflows before releasing new skills.
    \item Pair offensive simulations with SOC runbooks.
    \item Close the loop: feed discoveries back into Cerebro rules and memory hygiene.
  \end{itemize}
\end{frame}
